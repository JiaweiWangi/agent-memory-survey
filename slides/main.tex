% ============ 第 5 页（0:40）核心问题与边界 ============
\begin{frame}{长期记忆的核心问题与研究边界}
\begin{itemize}
  \item 长上下文 $\neq$ 记忆：会话内召回，无跨会话持久写入
  \item 静态 RAG $\neq$ 记忆：单查询检索，无读写闭环
  \item 参数知识 $\neq$ 记忆：推理时不可精确更新
  \item 五个研究问题：RQ1 定义边界 · RQ2 组织使用 · RQ3 更新遗忘 ·
        RQ4 可靠性 · RQ5 评测
\end{itemize}
\end{frame}

% ============ 第 6 页（1:20）研究脉络 ============
\begin{frame}{研究脉络与主要方法类别}
\textbf{检索到日志的演进}（谱系图，边=文献明确引用关系）
\begin{itemize}
  \item 2023 奠基：记忆流+三因子检索（Generative Agents）· 上下文工程
        （MemGPT 分页/中断）· 语言化经验（Reflexion/ExpeL）
  \item 2024–25 组织化：结构化图（A-Mem/HippoRAG 思路）· 时间知识图（Zep）·
        记忆操作（Mem0 ADD/UPDATE/DELETE/NOOP）
  \item 2026 前沿：遗忘治理（选择性遗忘/When to Forget）·
        可靠性（provenance/信任分数）· 效率（递归巩固/率失真压缩）
\end{itemize}
\end{frame}

% ============ 第 7 页（1:10）主要发现 ============
\begin{frame}{记忆更新、冲突与可靠性的主要发现}
\begin{itemize}
  \item \textbf{遗忘有效且便宜}：图记忆一次剪枝移除 9.8\% 节点/9.5\% 存储，
        token F1 变化 +0.001（CI 含 0）——但``图结构优于向量''被证伪
        （LongMemEval 上 F1 0.417 vs 0.468）
  \item \textbf{检索到 ≠ 用得上}：代理仅 55\% 时间对正确检索值采取行动，
        忽略率 0.45–0.53
  \item \textbf{错误会滚雪球}：记忆幻觉（HaluMem）与错误传播
        （状态转移雪球）是独立于攻击的可靠性问题
  \item \textbf{安全有代价}：记忆防御在良性场景损失效用并增加 token 开销
\end{itemize}
\end{frame}

% ============ 第 8 页（0:30）评测缺口与结论 ============
\begin{frame}{评测缺口、主要挑战和结论}
\begin{itemize}
  \item 评测缺口：LoCoMo 94\% 问题仅需 ≤2 个会话；``更新正确性''与
        ``何时记忆真正有帮助''缺统一基准
  \item 挑战：时效与冲突处理 · 错误传播的定量刻画 · 成本-收益报告不统一
  \item 结论：长期记忆研究正从``存得下''转向``信得过''——更新正确性与
        错误控制是核心开放问题
\end{itemize}
\end{frame}
